$$
	\int^\infty_{1} \frac{dx}{x^p}=\int_{1}^\infty x^{-p}dx;I(x)=\frac{1}{1-p}x^{-p-1}
$$
$$
	\left( \lim_{ \lambda \to \infty } \frac{1}{1-p}\lambda^{-p-1} \right) - \frac{1}{1-p}
$$
Bei p<1:
$$
	\lim_{ \lambda \to \infty } \frac{1}{1-p}\lambda^{-p+1}=\infty
$$

$$
	\int_{0}^1 \frac{dx}{x^p}=\int_{\lambda}^1 x^{-P} dx
$$
$$
	F(x)=\frac{1}{1-p}x^{1-p};F(1)=\frac{1}{1-p}
$$
$$
	\lim_{ \lambda \to 0 } \left( \frac{1}{1-p}\lambda^{1-p} - \frac{1}{1-p} \right)
$$
Bei p<1
$$
	\lim_{ \lambda \to 0 } \lambda^{1-p}=0
$$
Bei p>1
$$
	\lim_{ \lambda \to 0 } \lambda^{1-p}=\frac{1}{\lambda^{p-1}}=\infty
$$
Bei p=1
$$
	\lim_{ \lambda \to 0 } \lambda^{1-1}=1
